%macro READ 2 
    mov rax, 00 
    mov rdi, 00 
    mov rsi, %1 
    mov rdx, %2 
    syscall 
%endmacro 
 
%macro WRITE 2 
    mov rax, 01 
    mov rdi, 01 
    mov rsi, %1 
    mov rdx, %2 
    syscall 
%endmacro 
 
section .data 
    menu db 10, "1. BCD to 
HEX" 
         db 10, "2. HEX to BCD" 
         db 10, "3. EXITT" 
         db 10, "Enter choice " 
    menulen equ $-menu 
 
    msg1 db 10, "Enter BCD 
Number " 
    len1 equ $-msg1 
 
    msg2 db 10, "Enter HEX 
Number " 
    len2 equ $-msg2 
 
    msg3 db 10, "HEX 
Equivalent is " 
    len3 equ $-msg3 
 
    msg4 db 10, "BCD 
Equivalent is " 
    len4 equ $-msg4 
 
    msg5 db 10, "INVALID 
CHOICE" 
    len5 equ $-msg5 
 
section .bss 
    ans       resq 1 
    num       resq 1 
    char_buff resb 17 
    actl      resq 1 
    choice    resb 2 
    x         resb 1 
    cnt       resb 1 
 
section .text 
    global _start 
 
_start: 
    WRITE menu, menulen 
    READ choice, 02 
 
    cmp byte [choice], 31H 
    je bcd2hex 
 
    cmp byte [choice], 32H 
    je hex2bcd 
 
    cmp byte [choice], 33H 
    je exit 
 
    WRITE msg5, len5 
    jmp _start 
 
bcd2hex: 
    WRITE msg1, len1 
    READ char_buff, 17 
    dec rax 
    mov [actl], rax 
 
    mov rsi, char_buff 
    mov rax, 00H 
    mov rbx, 0AH 
 
l1: 
    mul rbx 
    mov rdx, 00H 
    mov dl, byte [rsi] 
    sub dl, 30H 
    add rax, rdx 
    inc rsi 
    dec qword [actl] 
    jnz l1 
 
    mov [ans], rax 
    WRITE msg3, len3 
    mov rbx, [ans] 
    call display 
    jmp _start 
 
hex2bcd: 
    WRITE msg2, len2 
    call accept 
    mov [num], rbx 
    mov byte [cnt], 00H 
 
    mov rbx, 0AH 
    mov rax, [num] 
 
l2: 
    mov rdx, 00H 
    div rbx 
    push rdx 
    inc byte [cnt] 
    cmp rax, 00H 
    jne l2 
 
    WRITE msg4, len4 
 
l3: 
    pop rdx 
    add dl, 30H 
    mov byte [x], dl 
    WRITE x, 01 
    dec byte [cnt] 
    jnz l3 
 
    jmp _start 
 
accept: 
    READ char_buff, 17 
    dec rax 
    mov qword [actl], rax 
 
    mov rsi, char_buff 
    mov rbx, 00H 
 
up: 
    shl rbx, 04H 
    mov rdx, 00H 
    mov dl, byte [rsi] 
    cmp dl, 39H 
    jbe sub_digit 
    sub dl, 07H 
 
sub_digit: 
    sub dl, 30H 
    add rbx, rdx 
    inc rsi 
    dec qword [actl] 
    jnz up 
    ret 
 
display: 
    mov rcx, 16 
    mov rsi, char_buff 
 
up2: 
    rol rbx, 04H 
    mov dl, bl 
    and dl, 0FH 
    cmp dl, 09H 
    jbe add_digit 
    add dl, 07H 
 
add_digit: 
    add dl, 30H 
    mov byte [rsi], dl 
    inc rsi 
    dec rcx 
    jnz up2 
 
    WRITE char_buff, 16 
    ret 
 
exit: 
    mov rax, 60 
    mov rdi, 00 
    syscall
